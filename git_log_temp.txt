+      let processedCount = 0;
+      filesToProcess.forEach(function(file) {
+        const reader = new FileReader();
+        reader.onload = function(evt) {
+          const img = new Image();
+          img.onload = function() {
+            const maxDim = 1280;
+            let w = img.width, h = img.height;
+            if (w > maxDim || h > maxDim) {
+              if (w > h) { h = Math.round((h * maxDim) / w); w = maxDim; }
+              else { w = Math.round((w * maxDim) / h); h = maxDim; }
+            }
+            const canvas = document.createElement('canvas');
+            canvas.width = w; canvas.height = h;
+            const ctx = canvas.getContext('2d');
+            ctx.drawImage(img, 0, 0, w, h);
+            const base64 = canvas.toDataURL('image/jpeg', 0.8);
+
+            selectedPhotos.push(base64);
+            processedCount++;
+
+            if (processedCount === filesToProcess.length) {
+              renderPhotoPreviews();
+              showToast('사진이 첨부되었습니다! 📸');
+            }
+          };
+          img.src = evt.target.result;
+        };
+        reader.readAsDataURL(file);
+      });

+      e.target.value = '';

+    function submitComment(e, rowIndex) {
+      e.preventDefault();
+      const authorInput = document.getElementById('comment-author-' + rowIndex);
+      const textInput = document.getElementById('comment-text-' + rowIndex);
+      const submitBtn = document.getElementById('comment-btn-' + rowIndex);
+
+      const author = authorInput.value.trim() || '작성자';
+      const text = textInput.value.trim();
+      if (!text) return;
+
+      localStorage.setItem('tennis_author_name', author);
+      savedAuthor = author;
+
+      submitBtn.disabled = true;
+      submitBtn.textContent = '...';
+
+      google.script.run
+        .withSuccessHandler(function(res) {
+          submitBtn.disabled = false;
+          submitBtn.textContent = '등록';
+          textInput.value = '';
+          showToast('댓글이 등록되었습니다! 💬');
+          
+          const log = allLogs.find(function(l) { return l.rowIndex === rowIndex; });
+          if (log) {
+            log.comments = res.comments;
+            renderLogs();
+          }
+        })
+        .withFailureHandler(function(err) {
+          submitBtn.disabled = false;
+          submitBtn.textContent = '등록';
+          alert('댓글 등록 실패: ' + err.message);
+        })
+        .addTennisComment(rowIndex, { author: author, text: text });
+    }

+    // ✏️ 댓글 수정
+    function editComment(rowIndex, commentId, safeText) {
+      const currentText = decodeURIComponent(safeText || '');
+      const newText = prompt('댓글/피드백을 수정하세요:', currentText);
+      if (newText === null || newText.trim() === '' || newText.trim() === currentText) return;
+
+      showToast('댓글 수정 중...');
+      google.script.run
+        .withSuccessHandler(function(res) {
+          showToast('댓글이 수정되었습니다! ✏️');
+          const log = allLogs.find(function(l) { return l.rowIndex === rowIndex; });
+          if (log) {
+            log.comments = res.comments;
+            renderLogs();
+          }
+        })
+        .withFailureHandler(function(err) {
+          alert('댓글 수정 실패: ' + err.message);
+        })
+        .updateTennisComment(rowIndex, commentId, newText.trim());
+    }

+          if (log) {

+        })

+          alert('댓글 삭제 실패: ' + err.message);
+        })
+        .deleteTennisComment(rowIndex, commentId);

+      return String(str)
+        .replace(/&/g, '&amp;')
+        .replace(/</g, '&lt;')
+        .replace(/>/g, '&gt;')
+        .replace(/"/g, '&quot;')
